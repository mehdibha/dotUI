## Installation

```npm
npx shadcn@latest add @dotui/skeleton
```

## Usage

```tsx
import { Skeleton } from "@/components/ui/skeleton"
```

```tsx
<Skeleton isLoading={true}>
  <Skeleton className="h-4 w-32" />
</Skeleton>
```
